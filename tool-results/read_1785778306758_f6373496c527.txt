   363→function getInitials(name: string | null | undefined): string {
   364→  if (!name) return "?"
   365→  return name.trim().split(/\s+/).slice(0, 2).map(s => s[0]?.toUpperCase() || "").join("") || "?"
   366→}
   367→
   368→function parseAddOns(raw: any): AddOn[] {
   369→  if (!raw) return []
   370→  if (Array.isArray(raw)) return raw as AddOn[]
   371→  if (typeof raw === "string") {
   372→    try {
   373→      const parsed = JSON.parse(raw)
   374→      return Array.isArray(parsed) ? parsed : []
   375→    } catch {
   376→      return []
   377→    }
   378→  }
   379→  return []
   380→}
   381→
   382→function parseDetails(raw: any): any {
   383→  if (raw == null) return null
   384→  if (typeof raw === "string") {
   385→    try { return JSON.parse(raw) } catch { return null }
   386→  }
   387→  return raw
   388→}

   390→function isSameDay(a: Date, b: Date): boolean {
   391→  return a.getFullYear() === b.getFullYear() &&
   392→    a.getMonth() === b.getMonth() &&
   393→    a.getDate() === b.getDate()
   394→}

   396→function slotDateKey(slot: Slot): string {
   397→  return new Date(slot.date).toISOString().slice(0, 10)
   398→}

   403→export default function BookingsView() {
   404→  const setView = useApp((s) => s.setView)
   405→  const setSelectedConversation = useApp((s) => s.setSelectedConversation)
   406→  const currentStaffId = useApp((s) => s.currentStaffId)
   407→
   408→  const [orders, setOrders] = useState<Order[]>([])
   409→  const [allOrders, setAllOrders] = useState<Order[]>([])
   410→  const [loading, setLoading] = useState(true)
   411→  const [error, setError] = useState<string | null>(null)
   412→
   413→  // Filters
   414→  const [statusTab, setStatusTab] = useState<StatusKey>("ALL")
   415→  const [channel, setChannel] = useState<string>("ALL")
   416→  const [search, setSearch] = useState("")
   417→  const [searchInput, setSearchInput] = useState("")
   418→  const [dateFrom, setDateFrom] = useState<Date | undefined>(undefined)
   419→  const [dateTo, setDateTo] = useState<Date | undefined>(undefined)
   420→  const [datePopoverOpen, setDatePopoverOpen] = useState(false)
   421→
   422→  // Dialogs
   423→  const [detailOrder, setDetailOrder] = useState<Order | null>(null)
   424→  const [newBookingOpen, setNewBookingOpen] = useState(false)
   425→  const [rescheduleTarget, setRescheduleTarget] = useState<Order | null>(null)
   426→  const [cancelTarget, setCancelTarget] = useState<Order | null>(null)
   427→
   428→  // ---- fetch orders (server-side filters: status/channel/search) ----
   429→  const fetchOrders = useCallback(async () => {
   430→    try {
   431→      const params = new URLSearchParams()
   432→      if (statusTab !== "ALL") params.set("status", statusTab)
   433→      if (channel !== "ALL") params.set("channel", channel)
   434→      if (search) params.set("search", search)
   435→      params.set("limit", "200")
   436→      const url = `/api/orders${params.toString() ? "?" + params.toString() : ""}`
   437→      const res = await fetch(url, { cache: "no-store" })
   438→      if (!res.ok) throw new Error("Failed to load orders")
   439→      const json = await res.json()
   440→      setOrders(json.orders || [])
   441→      setError(null)
   442→    } catch (e: any) {
   443→      setError(e?.message || "Unable to load orders")
   444→    } finally {
   445→      setLoading(false)
   446→    }
   447→  }, [statusTab, channel, search])
   448→
   449→  // ---- fetch all orders for stats ----
   450→  const fetchAllOrders = useCallback(async () => {
   451→    try {
   452→      const res = await fetch("/api/orders?limit=500", { cache: "no-store" })
   453→      if (!res.ok) return
   454→      const json = await res.json()
   455→      setAllOrders(json.orders || [])
   456→    } catch {
   457→      // silent — stats are best-effort
   458→    }
   459→  }, [])
   460→
   461→  useEffect(() => {
   462→    setLoading(true)
   463→    fetchOrders()
   464→  }, [fetchOrders])
   465→
   466→  useEffect(() => {
   467→    fetchAllOrders()
   468→  }, [fetchAllOrders])
   469→
   470→  // auto-refresh every 20s
   471→  useEffect(() => {
   472→    const t = setInterval(() => {
   473→      fetchOrders()
   474→      fetchAllOrders()
   475→    }, 20_000)
   476→    return () => clearInterval(t)
   477→  }, [fetchOrders, fetchAllOrders])
   478→
   479→  // ---- search submit (debounced via Enter / button) ----
   480→  const submitSearch = () => {
   481→    setSearch(searchInput.trim())
   482→  }
   483→
   484→  // ---- derived stats (from full dataset) ----
   485→  const stats = useMemo(() => {
   486→    return {
   487→      total: allOrders.length,
   488→      confirmed: allOrders.filter(o => o.orderStatus === "CONFIRMED").length,
   489→      pendingPayment: allOrders.filter(o => o.orderStatus === "PENDING_PAYMENT").length,
   490→      completed: allOrders.filter(o => o.orderStatus === "COMPLETED").length,
   491→      cancelled: allOrders.filter(o => o.orderStatus === "CANCELLED").length,
   492→    }
   493→  }, [allOrders])
   494→
   495→  // ---- client-side date filtering on already-fetched orders ----
   496→  const visibleOrders = useMemo(() => {
   497→    return orders.filter(o => {
   498→      const created = new Date(o.createdAt)
   499→      if (dateFrom && created < dateFrom) return false
   500→      if (dateTo) {
   501→        const end = new Date(dateTo)
   502→        end.setHours(23, 59, 59, 999)
   503→        if (created > end) return false
   504→      }
   505→      return true
   506→    })
   507→  }, [orders, dateFrom, dateTo])
   508→
   509→  const hasActiveFilters = statusTab !== "ALL" || channel !== "ALL" || !!search || !!dateFrom || !!dateTo
   510→
   511→  const clearFilters = () => {
   512→    setStatusTab("ALL")
   513→    setChannel("ALL")
   514→    setSearch("")
   515→    setSearchInput("")
   516→    setDateFrom(undefined)
   517→    setDateTo(undefined)
   518→  }
   519→
   520→  // ---- generic order action ----
   521→  const handleOrderAction = async (
   522→    order: Order,
   523→    action: "CANCEL" | "COMPLETE" | "NO_SHOW" | "RESCHEDULE",
   524→    extra: Record<string, any> = {}
   525→  ) => {
   526→    const tid = `act-${order.id}-${action}`
   527→    toast.loading("Updating order…", { id: tid })
   528→    try {
   529→      const res = await fetch(`/api/orders/${order.id}`, {
   530→        method: "PATCH",
   531→        headers: { "Content-Type": "application/json" },
   532→        body: JSON.stringify({ action, staffId: currentStaffId, ...extra }),
   533→      })
   534→      if (!res.ok) {
   535→        const j = await res.json().catch(() => ({}))
   536→        throw new Error(j.error || "Action failed")
   537→      }
   538→      const data = await res.json()
   539→      toast.success(`Order ${action.toLowerCase().replace("_", " ")} succeeded`, { id: tid, icon: <CheckCircle2 className="h-4 w-4" /> })
   540→      // refresh
   541→      fetchOrders()
   542→      fetchAllOrders()
   543→      // refresh detail if open
   544→      if (detailOrder?.id === order.id) {
   545→        setDetailOrder(data.order)
   546→      }
   547→      return true
   548→    } catch (e: any) {
   549→      toast.error(e?.message || "Could not update order", { id: tid })
   550→      return false
   551→    }
   552→  }
   553→
   554→  // ---- approve payment (uses payments API) ----
   555→  const handleApprovePayment = async (order: Order) => {
   556→    const submitted = (order.payments || []).find(p => p.status === "SUBMITTED" || p.status === "PENDING")
   557→    if (!submitted) {
   558→      toast.error("No pending payment found for this order")
   559→      return
   560→    }
   561→    const tid = `appr-${order.id}`
   562→    toast.loading("Approving payment…", { id: tid })
   563→    try {
   564→      const res = await fetch(`/api/payments/${submitted.id}/verify`, {
   565→        method: "POST",
   566→        headers: { "Content-Type": "application/json" },
   567→        body: JSON.stringify({ action: "APPROVE", staffId: currentStaffId, comment: "Approved from bookings view" }),
   568→      })
   569→      if (!res.ok) {
   570→        const j = await res.json().catch(() => ({}))
   571→        throw new Error(j.error || "Approve failed")
   572→      }
   573→      const data = await res.json()
   574→      toast.success(
   575→        `Payment approved — ${data.voucher?.voucherCode ? "Voucher " + data.voucher.voucherCode + " issued" : "order confirmed"}`,
   576→        { id: tid, icon: <CheckCircle2 className="h-4 w-4" /> }
   577→      )
   578→      fetchOrders()
   579→      fetchAllOrders()
   580→      // refresh detail
   581→      const detail = await fetch(`/api/orders/${order.id}`, { cache: "no-store" }).then(r => r.json()).catch(() => null)
   582→      if (detail?.order) setDetailOrder(detail.order)
   583→    } catch (e: any) {
   584→      toast.error(e?.message || "Could not approve payment", { id: tid })
   585→    }
   586→  }
   587→
   588→  // ---- open WhatsApp conversation ----
   589→  const openWhatsApp = async (order: Order) => {
   590→    try {
   591→      toast.loading("Looking up conversation…", { id: `wa-${order.id}` })
