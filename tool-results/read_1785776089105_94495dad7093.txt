   364→function parseJsonArray<T>(raw: any, fallback: T[] = []): T[] {
   365→  if (!raw) return fallback
   366→  if (Array.isArray(raw)) return raw as T[]
   367→  if (typeof raw === "string") {
   368→    try {
   369→      const parsed = JSON.parse(raw)
   370→      return Array.isArray(parsed) ? parsed : fallback
   371→    } catch {
   372→      return fallback
   373→    }
   374→  }
   375→  return fallback
   376→}

   378→function slugify(s: string): string {
   379→  return s
   380→    .toLowerCase()
   381→    .trim()
   382→    .replace(/[^a-z0-9\s-]/g, "")
   383→    .replace(/\s+/g, "-")
   384→    .replace(/-+/g, "-")
   385→    .replace(/^-|-$/g, "")
   386→}

   388→function isoDate(d: Date): string {
   389→  // YYYY-MM-DD in local timezone
   390→  const y = d.getFullYear()
   391→  const m = String(d.getMonth() + 1).padStart(2, "0")
   392→  const day = String(d.getDate()).padStart(2, "0")
   393→  return `${y}-${m}-${day}`
   394→}
   395→
   396→function dateKey(d: Date | string): string {
   397→  const date = typeof d === "string" ? new Date(d) : d
   398→  return isoDate(date)
   399→}
   400→
   401→function isSameDay(a: Date, b: Date): boolean {
   402→  return a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate()
   403→}
   404→
   405→function isWeekend(d: Date): boolean {
   406→  // Oman weekend: Friday (5) + Saturday (6)
   407→  return d.getDay() === 5 || d.getDay() === 6
   408→}

   410→function getInitials(name?: string | null): string {
   411→  if (!name) return "?"
   412→  const parts = name.trim().split(/\s+/).slice(0, 2)
   413→  return parts.map(p => p[0]?.toUpperCase() || "").join("") || "?"
   414→}

   416→function ratingStars(rating: number): { full: number; half: number; empty: number } {
   417→  const full = Math.floor(rating)
   418→  const decimal = rating - full
   419→  const half = decimal >= 0.25 && decimal < 0.75 ? 1 : 0
   420→  const extra = decimal >= 0.75 ? 1 : 0
   421→  return { full: full + extra, half, empty: 5 - full - extra - half }
   422→}

  1649→function TourFormDialog({
  1650→  open,
  1651→  onOpenChange,
  1652→  initialTour,
  1653→  onSaved,
  1654→}: {
  1655→  open: boolean
  1656→  onOpenChange: (o: boolean) => void
  1657→  initialTour: Tour | null
  1658→  onSaved: (t: Tour) => void
  1659→}) {
  1660→  const [form, setForm] = useState<FormState>(emptyFormState())
  1661→  const [saving, setSaving] = useState(false)
  1662→  const [autoSlug, setAutoSlug] = useState(true)
  1663→
  1664→  // Reset form when target changes (use key on parent for full reset)
  1665→  useEffect(() => {
  1666→    if (open) {
  1667→      setForm(initialTour ? formFromTour(initialTour) : emptyFormState())
  1668→      setAutoSlug(!initialTour)

  1671→
  1672→  const isEdit = !!initialTour
  1673→
  1674→  // Auto-slug from name
  1675→  useEffect(() => {
  1676→    if (autoSlug && form.name) {
  1677→      setForm(prev => ({ ...prev, slug: slugify(prev.name) }))
  1678→    }
  1679→  }, [form.name, autoSlug])
  1680→
  1681→  const update = useCallback(<K extends keyof FormState>(key: K, value: FormState[K]) => {
  1682→    setForm(prev => ({ ...prev, [key]: value }))
  1683→  }, [])
  1684→
  1685→  const handleSubmit = async () => {
  1686→    if (!form.name.trim()) {
  1687→      toast.error("Tour name is required")
  1688→      return
  1689→    }
  1690→    if (!form.slug.trim()) {
  1691→      toast.error("Slug is required")
  1692→      return
  1693→    }
  1694→    if (!form.description.trim()) {
  1695→      toast.error("Description is required")
  1696→      return
  1697→    }
  1698→
  1699→    const payload: any = {
  1700→      slug: form.slug,
  1701→      name: form.name,
  1702→      nameAr: form.nameAr || null,
  1703→      description: form.description,
  1704→      category: form.category,
  1705→      city: form.city,
  1706→      basePrice: Number(form.basePrice) || 0,
  1707→      childPrice: form.childPrice === "" ? null : Number(form.childPrice),
  1708→      groupPrice: form.groupPrice === "" ? null : Number(form.groupPrice),
  1709→      durationHours: Number(form.durationHours) || 0,
  1710→      difficulty: form.difficulty,
  1711→      capacityPerSlot: Number(form.capacityPerSlot) || 1,
  1712→      meetingPoint: form.meetingPoint || null,
  1713→      cancellationPolicy: form.cancellationPolicy || null,
  1714→      featured: form.featured,
  1715→      status: form.status,
  1716→      media: form.media,
  1717→      itinerary: form.itinerary,
  1718→      inclusions: form.inclusions,
  1719→      exclusions: form.exclusions,
  1720→      whatToBring: form.whatToBring,
  1721→      addOns: form.addOns.map(a => ({ name: a.name, price: Number(a.price) || 0, type: a.type, isActive: a.isActive !== false })),
  1722→    }
  1723→
  1724→    setSaving(true)
  1725→    try {
  1726→      const url = isEdit ? `/api/tours/${initialTour!.id}` : "/api/tours"
  1727→      const method = isEdit ? "PUT" : "POST"
  1728→      const res = await fetch(url, {
  1729→        method,
  1730→        headers: { "Content-Type": "application/json" },
  1731→        body: JSON.stringify(payload),
  1732→      })
  1733→      if (!res.ok) {
  1734→        const err = await res.json().catch(() => ({}))
  1735→        throw new Error(err.error || `Failed to ${isEdit ? "update" : "create"} tour`)
  1736→      }
  1737→      const json = await res.json()
  1738→      onSaved(json.tour)
  1739→    } catch (e: any) {
  1740→      toast.error(`Failed to ${isEdit ? "update" : "create"} tour`, { description: e?.message })
  1741→    } finally {
  1742→      setSaving(false)
  1743→    }
  1744→  }
  1745→
  1746→  return (
  1747→    <Dialog open={open} onOpenChange={onOpenChange}>
  1748→      <DialogContent className="max-h-[92vh] w-full max-w-4xl overflow-hidden p-0 sm:max-w-4xl">
  1749→        <DialogHeader className="border-b border-stone-200 bg-white px-6 py-4">
  1750→          <DialogTitle className="flex items-center gap-2 text-lg">
  1751→            {isEdit ? <Pencil className="size-5 text-emerald-600" /> : <Plus className="size-5 text-emerald-600" />}
  1752→            {isEdit ? "Edit Tour" : "Add New Tour"}
  1753→          </DialogTitle>
  1754→          <DialogDescription>
  1755→            {isEdit
  1756→              ? "Update the tour details below."
  1757→              : "Fill in the tour details below to add it to your catalogue."}
  1758→          </DialogDescription>
  1759→        </DialogHeader>
  1760→
  1761→        <FormTabs form={form} update={update} autoSlug={autoSlug} setAutoSlug={setAutoSlug} />
  1762→
  1763→        <DialogFooter className="border-t border-stone-200 bg-stone-50 px-6 py-3">
  1764→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving}>
  1765→            Cancel
  1766→          </Button>
  1767→          <Button
  1768→            onClick={handleSubmit}
  1769→            disabled={saving}
  1770→            className="bg-gradient-to-r from-emerald-600 to-teal-600 text-white hover:from-emerald-700 hover:to-teal-700"
  1771→          >
  1772→            {saving ? <Loader2 className="size-4 animate-spin" /> : <Save className="size-4" />}
  1773→            {isEdit ? "Save Changes" : "Create Tour"}
  1774→          </Button>
  1775→        </DialogFooter>
  1776→      </DialogContent>
  1777→    </Dialog>
